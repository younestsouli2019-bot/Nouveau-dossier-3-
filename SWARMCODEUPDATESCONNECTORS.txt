export class IdempotencyTracker {
  constructor(storagePath = './.idempotency_store.jsonl') {
    this.storagePath = path.resolve(storagePath);
  }

  generateKey(payload) {
    if (!payload) throw new Error('Payload is required to generate an idempotency key.');
    const serialized = typeof payload === 'string' ? payload : JSON.stringify(payload);
    return crypto.createHash('sha256').update(serialized).digest('hex');
  }

  isDuplicate(key) {
    if (!fs.existsSync(this.storagePath)) return false;
    try {
      const logData = fs.readFileSync(this.storagePath, 'utf8');
      const entries = logData.split('\n').filter(Boolean);
      return entries.some(line => {
        try {
          return JSON.parse(line).key === key;
        } catch {
          return false;
        }
      });
    } catch (error) {
      console.error(`[Idempotency] Failed reading log store: ${error.message}`);
      return false;
    }
  }

  saveKey(key, responseMetadata = {}) {
    const logEntry = {
      key,
      timestamp: new Date().toISOString(),
      meta: responseMetadata
    };
    try {
      fs.appendFileSync(this.storagePath, JSON.stringify(logEntry) + '\n', 'utf8');
    } catch (error) {
      console.error(`[Idempotency] Failed recording token key ${key}: ${error.message}`);
    }
  }
}

import path from 'path';

export class DeadLetterQueue {
  constructor(dlqPath = './.dlq_store.jsonl') {
    this.dlqPath = path.resolve(dlqPath);
  }

  enqueue(failedPayload, errorReason = 'Unknown execution exception') {
    const dlqRecord = {
      id: `dlq_${Date.now()}_${crypto.randomBytes(3).toString('hex')}`,
      timestamp: new Date().toISOString(),
      payload: failedPayload,
      reason: errorReason
    };
    try {
      fs.appendFileSync(this.dlqPath, JSON.stringify(dlqRecord) + '\n', 'utf8');
      console.warn(`[DLQ] Message isolated inside fallback store: ${dlqRecord.id}`);
    } catch (error) {
      console.error(`[DLQ Critical failure] Cannot isolate payload: ${error.message}`);
    }
  }

  getStats() {
    if (!fs.existsSync(this.dlqPath)) return { count: 0, sizeBytes: 0 };
    try {
      const stats = fs.statSync(this.dlqPath);
      const lines = fs.readFileSync(this.dlqPath, 'utf8').split('\n').filter(Boolean);
      return { count: lines.length, sizeBytes: stats.size };
    } catch (error) {
      return { error: `Unreadable ledger payload: ${error.message}` };
    }
  }

  peek(limit = 10) {
    if (!fs.existsSync(this.dlqPath)) return [];
    try {
      const lines = fs.readFileSync(this.dlqPath, 'utf8').split('\n').filter(Boolean);
      return lines.slice(0, limit).map(line => JSON.parse(line));
    } catch (error) {
      return [{ error: `Failed to peek items: ${error.message}` }];
    }
  }

  purge() {
    try {
      if (fs.existsSync(this.dlqPath)) {
        fs.unlinkSync(this.dlqPath);
        return true;
      }
    } catch (error) {
      console.error(`[DLQ Error] Failed purging file log: ${error.message}`);
    }
    return false;
  }
}

export class IPAllowlist {
  constructor(allowedIpsEnv = process.env.WEBHOOK_ALLOWED_IPS || '') {
    this.cidrs = [];
    if (!allowedIpsEnv.trim()) {
      console.warn('[IP Allowlist] WEBHOOK_ALLOWED_IPS empty. Webhook access guard disabled.');
      return;
    }
    
    this.cidrs = allowedIpsEnv.split(',')
                              .map(s => s.trim())
                              .filter(Boolean)
                              .map(cidr => {
                                try {
                                  return ipaddr.parseCIDR(cidr.includes('/') ? cidr : `${cidr}/32`);
                                } catch (e) {
                                  console.error(`[IP Allowlist] Invalid configuration block format "${cidr}":`, e.message);
                                  return null;
                                }
                              }).filter(Boolean);
  }

  isAllowed(clientIp) {
    if (this.cidrs.length === 0) return true;
    if (!clientIp) return false;
    
    try {
      let cleanedIp = clientIp;
      if (ipaddr.IPv6.isValid(clientIp)) {
        const addr = ipaddr.IPv6.parse(clientIp);
        if (addr.isIPv4MappedAddress()) {
          cleanedIp = addr.toIPv4Address().toString();
        }
      }
      
      const parsedClient = ipaddr.process(cleanedIp);
      return this.cidrs.some(cidr => parsedClient.match(cidr));
    } catch (err) {
      console.error(`[IP Allowlist] Error parsing client connection block "${clientIp}":`, err.message);
      return false;
    }
  }
}
utils/webhook-dedup.mjsjavascriptexport class WebhookDedup {
  constructor(maxEntries = 100000, ttlDays = 7) {
    this.cache = new Map();
    this.maxEntries = maxEntries;
    this.ttlMs = ttlDays * 24 * 60 * 60 * 1000;
  }

  isDuplicate(eventId) {
    if (!eventId) return false;
    this.cleanup();

    if (this.cache.has(eventId)) {
      this.cache.set(eventId, Date.now()); // Bump activity access flag
      return true;
    }

    if (this.cache.size >= this.maxEntries) {
      const oldestKey = this.cache.keys().next().value;
      this.cache.delete(oldestKey);
    }

    this.cache.set(eventId, Date.now());
    return false;
  }

  cleanup() {
    const now = Date.now();
    let count = 0;
    for (const [key, value] of this.cache.entries()) {
      if (now - value > this.ttlMs) {
        this.cache.delete(key);
        count++;
      }
    }
    if (count > 0) {
      console.log(`[Webhook Dedup] Swept ${count} expired logs out of transaction memory context.`);
    }
  }
}
utils/distributed-lock.mjsjavascriptimport fs from 'fs';
import path from 'path';

export class DistributedLock {
  constructor(lockName, lockDir = './.locks') {
    this.lockDir = path.resolve(lockDir);
    this.lockPath = path.join(this.lockDir, `${lockName}.lock`);
  }

  async acquire(ttlMs = 30000) {
    try {
      if (!fs.existsSync(this.lockDir)) {
        fs.mkdirSync(this.lockDir, { recursive: true });
      }

      if (fs.existsSync(this.lockPath)) {
        const stat = fs.statSync(this.lockPath);
        const age = Date.now() - stat.mtimeMs;
        
        if (age < ttlMs) {
          return false; // Active lock held
        }
        this.release(); // Evict expired lock
      }

      const lockPayload = { pid: process.pid, acquiredAt: Date.now() };
      fs.writeFileSync(this.lockPath, JSON.stringify(lockPayload), 'utf8');
      return true;
    } catch (error) {
      console.error(`[Distributed Lock] Resource claim error: ${error.message}`);
      return false;
    }
  }

  release() {
    try {
      if (fs.existsSync(this.lockPath)) {
        fs.unlinkSync(this.lockPath);
      }
    } catch (error) {
      // Ignore if already deleted by a racing thread
    }
  }
}

export async function base44Request(endpoint, payload, config = {}) {
  const url = `${config.baseUrl || 'https://base44.io'}${endpoint}`;
  const method = config.method || 'POST';
  const headers = {
    'Content-Type': 'application/json',
    'X-Client-Version': '2.2.0',
    ...config.headers
  };

  if (config.secret) {
    const timestamp = Date.now().toString();
    const payloadStr = payload ? JSON.stringify(payload) : '';
    const signature = crypto.createHmac('sha256', config.secret)
                            .update(timestamp + payloadStr)
                            .digest('hex');
    headers['X-B44-Signature'] = signature;
    headers['X-B44-Timestamp'] = timestamp;
  }

  try {
    const response = await fetch(url, {
      method,
      headers,
      body: payload ? JSON.stringify(payload) : undefined,
      signal: config.timeout ? AbortSignal.timeout(config.timeout) : undefined
    });

    if (!response.ok) {
      const errText = await response.text().catch(() => 'No data body');
      throw new Error(`Base44 Remote Service Failure [${response.status}]: ${errText}`);
    }

    return await response.json();
  } catch (error) {
    throw new Error(`[Network Layer Connection Fault] ${error.message}`);
  }
}
