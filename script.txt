def run(cmd, cwd=None, env=None):
  printable = cmd if isinstance(cmd, str) else " ".join(str(x) for x in cmd)
  sys.stdout.write(f"$ {printable}\n")
  res = subprocess.run(cmd, cwd=cwd, env=env, shell=isinstance(cmd, str), capture_output=True, text=True)
  if res.stdout:
    sys.stdout.write(res.stdout)
    if not res.stdout.endswith("\n"):
      sys.stdout.write("\n")
  if res.stderr:
    sys.stderr.write(res.stderr)
    if not res.stderr.endswith("\n"):
      sys.stderr.write("\n")
  if res.returncode != 0:
    raise RuntimeError(f"Command failed ({res.returncode}): {printable}")
  return res


def try_parse_json_from_stdout(stdout_text: str):
  if not stdout_text:
    return None
  lines = [ln.strip() for ln in str(stdout_text).splitlines() if ln.strip()]
  if not lines:
    return None
  candidates = []
  if lines[-1].startswith("{") and lines[-1].endswith("}"):
    candidates.append(lines[-1])
  joined = "\n".join(lines)
  if joined.startswith("{") and joined.endswith("}"):
    candidates.append(joined)
  for c in candidates:
    try:
      return json.loads(c)
    except Exception:
      pass
  return None


def has_base44_online_creds(env):
  if env.get("BASE44_APP_ID") and env.get("BASE44_SERVICE_TOKEN"):
    return True
  if env.get("BASE44_API_KEY") or env.get("BASE44_API_TOKEN") or env.get("BASE44_KEY"):
    return True
  return False


def flatten_single_root(dir_path: Path):
  items = [p for p in dir_path.iterdir()]
  if len(items) != 1 or not items[0].is_dir():
    return
  root = items[0]
  for child in root.iterdir():
    shutil.move(str(child), str(dir_path))
  shutil.rmtree(root)


def find_zip_by_name(search_root: Path, name_substring: str):
  if not search_root.exists():
    return None
  for p in sorted(search_root.rglob("*.zip")):
    if name_substring in p.name:
      return p
  return None


def resolve_input_zip():
  from_env = os.environ.get("ZIP_PATH")
  if from_env:
    p = Path(from_env)
    if p.exists():
      return p
    raise FileNotFoundError(f"ZIP_PATH points to missing file: {p}")

  candidates = [
    Path("/content/Ensuring Deployment Success Before Free Tier Limit (3) (1).zip"),
    Path("/content/Ensuring Deployment Success Before Free Tier Limit (3).zip"),
    Path("/content/Ensuring Deployment Success Before Free Tier Limit.zip")
  ]
  for c in candidates:
    if c.exists():
      return c

  found = find_zip_by_name(Path("/content"), "Ensuring Deployment Success Before Free Tier Limit")
  if found:
    return found

  found = find_zip_by_name(Path("/content/drive/MyDrive"), "Ensuring Deployment Success Before Free Tier Limit")
  if found:
    return found

  raise FileNotFoundError(
    "Upload the zip to Colab (/content) or Google Drive, then rerun. "
    'Optionally set ZIP_PATH="/content/yourfile.zip".'
  )


def extract_zip(zip_path: Path, target_dir: Path):
  target_dir.mkdir(parents=True, exist_ok=True)
  with zipfile.ZipFile(zip_path, "r") as z:
    z.extractall(target_dir)
  flatten_single_root(target_dir)


def merge_nested_zip_if_present(project_dir: Path, nested_zip_name: str):
  nested = project_dir / nested_zip_name
  if not nested.exists():
    return False
  tmp = project_dir / f".tmp_extract_{re.sub(r'[^a-zA-Z0-9_]+', '_', nested_zip_name)}"
  if tmp.exists():
    shutil.rmtree(tmp)
  extract_zip(nested, tmp)
  for src in tmp.rglob("*"):
    rel = src.relative_to(tmp)
    dest = project_dir / rel
    if src.is_dir():
      dest.mkdir(parents=True, exist_ok=True)
    else:
      dest.parent.mkdir(parents=True, exist_ok=True)
      shutil.copy2(src, dest)
  shutil.rmtree(tmp)
  return True


def find_project_root(extracted_dir: Path):
  pkg_files = list(extracted_dir.rglob("package.json"))
  if not pkg_files:
    raise FileNotFoundError("package.json not found after extraction")
  pkg_files.sort(key=lambda p: len(p.parts))
  return pkg_files[0].parent


def ensure_node():
  ok = False
  try:
    out = subprocess.check_output(["node", "-v"], text=True).strip()
    major = int(out.lstrip("v").split(".")[0])
    ok = major >= 18
  except Exception:
    ok = False

  if ok:
    run(["node", "-v"])
    run(["npm", "-v"])
    return

  run("sudo apt-get update -y")
  run("sudo apt-get install -y ca-certificates curl gnupg")
  run("curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -")
  run("sudo apt-get install -y nodejs")
  run(["node", "-v"])
  run(["npm", "-v"])
