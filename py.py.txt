import base64
import hashlib
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import ec
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.backends import default_backend

# Inputs from the user's snippet
MESSAGE_STR = '9f7c9c9e1c7a8b5f6d3e2f1a0b9c8d7e6f5a4b3c2d1e0f9a8b7c6d5e4f3a2b1c0d'
SIGNATURE_B64 = 'MEUCIE4cPTbOI9dgm3CuOWomIV8ALM5G5ogYv49OHXeTPXvfAiEA30hMmS0sKYRPX7UrnFvd5OoCuq4FW2UA5lsd5UnTSEw='
PUBLIC_KEY_PEM = """
-----BEGIN PUBLIC KEY-----
MFkwEwYHKoZIzj0CAQYIKoZIzj0DAQcDQgAEoiY+uD4JB5ZWDXyB46kI7WEKJgaz
aAY7hr8VU5I7Y33XmxpUDwT2bucB+vZ3g48ts3EywSzp8/EttxzVne8U7A==
-----END PUBLIC KEY-----
"""

try:
    # 1. Load the Public Key
    public_key = serialization.load_pem_public_key(
        PUBLIC_KEY_PEM.encode(),
        backend=default_backend()
    )

    # 2. Decode the Signature
    signature_bytes = base64.b64decode(SIGNATURE_B64)

    # 3. Prepare the Message
    # Node.js crypto.verify.update(string) treats the string as UTF-8 and hashes it.
    # The signature was created over the SHA-256 hash of the UTF-8 bytes of the string.
    message_bytes = MESSAGE_STR.encode('utf-8')
    
    # 4. Verify
    # The cryptography library handles the hashing internally if we pass the signature verification function the data.
    # However, since the Node code uses SHA256 specifically:
    valid = public_key.verify(
        signature_bytes,
        message_bytes,
        ec.ECDSA(hashes.SHA256())
    )

    print(f'VALID: {valid}')

except Exception as e:
    print(f"Error during verification: {e}")
