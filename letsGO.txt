#### **A. The "Agentic SRE" Gas Guard (`scripts/check-gas-balance.mjs`)**
This is your first line of autonomous operational integrity.

```javascript
// scripts/check-gas-balance.mjs
import Web3 from 'web3'; // For EVM chains (BSC, Polygon, etc.)
import TronWeb from 'tronweb'; // For TRON
import dotenv from 'dotenv';
import { createRequire } from 'module';
const require = createRequire(import.meta.url);
const colors = require('colors');

dotenv.config();

// ===== CONFIGURATION (Machine-Readable Agent Policy) =====
const AGENT_GAS_POLICY = {
  // Define your operational wallet per chain
  WALLETS: {
    BSC: process.env.BSC_SETTLEMENT_WALLET,
    TRON: process.env.TRON_SETTLEMENT_WALLET,
    POLYGON: process.env.POLYGON_SETTLEMENT_WALLET,
    TON: process.env.TON_SETTLEMENT_WALLET, // May need custom RPC
  },
  // Minimum native token balance required to approve a route (in base units)
  MINIMUM_SAFE_BALANCE: {
    BSC: Web3.utils.toWei('0.01', 'ether'),      // ~0.01 BNB
    TRON: 20_000_000,                            // ~20 TRX (in sun)
    POLYGON: Web3.utils.toWei('0.5', 'ether'),   // ~0.5 MATIC
    TON: 1_000_000_000,                          // ~1 TON (in nanoton)
  },
  // Public RPC Endpoints (Fallback; consider using paid nodes for production)
  RPC_ENDPOINTS: {
    BSC: 'https://bsc-dataseed1.binance.org/',
    TRON: 'https://api.trongrid.io',
    POLYGON: 'https://polygon-rpc.com',
    TON: 'https://toncenter.com/api/v2/jsonRPC', // TON requires API key
  }
};

// ===== AGENT LOGIC: Balance Check =====
async function checkChainBalance(chain) {
  const wallet = AGENT_GAS_POLICY.WALLETS[chain];
  const minBalance = AGENT_GAS_POLICY.MINIMUM_SAFE_BALANCE[chain];
  const rpc = AGENT_GAS_POLICY.RPC_ENDPOINTS[chain];

  if (!wallet) {
    return { chain, status: 'CONFIG_MISSING', message: 'Wallet not configured in .env' };
  }

  try {
    let currentBalance;
    switch (chain) {
      case 'BSC':
      case 'POLYGON': {
        const web3 = new Web3(rpc);
        currentBalance = await web3.eth.getBalance(wallet);
        break;
      }
      case 'TRON': {
        const tronWeb = new TronWeb({ fullHost: rpc });
        const account = await tronWeb.trx.getAccount(wallet);
        currentBalance = account.balance || 0;
        break;
      }
      case 'TON': {
        // TON requires specialized library (@ton/ton) and often an API key
        // This is a placeholder structure
        const response = await fetch(rpc, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            jsonrpc: '2.0',
            id: 1,
            method: 'getAddressBalance',
            params: { address: wallet }
          })
        });
        const data = await response.json();
        currentBalance = data.result || 0;
        break;
      }
      default:
        return { chain, status: 'UNSUPPORTED', message: 'Chain logic not implemented' };
    }

    const isSufficient = BigInt(currentBalance) >= BigInt(minBalance);
    return {
      chain,
      wallet: wallet.substring(0, 8) + '...',
      currentBalance: currentBalance.toString(),
      minimumRequired: minBalance.toString(),
      status: isSufficient ? 'SUFFICIENT' : 'INSUFFICIENT',
      routeRecommendation: isSufficient ? 'ROUTE_APPROVED' : 'ROUTE_BLOCKED'
    };

  } catch (error) {
    return { chain, status: 'RPC_ERROR', message: `Network error: ${error.message}` };
  }
}

// ===== AGENT ORCHESTRATION: Run All Checks =====
async function runGasAudit() {
  console.log(colors.bold.cyan('\n🔍 [OWNER SETTLEMENT AGENT] - Gas Reserve Audit'));
  console.log(colors.gray('Executing Agentic SRE pre-flight check...\n'));

  const chains = Object.keys(AGENT_GAS_POLICY.WALLETS);
  const results = await Promise.all(chains.map(checkChainBalance));

  // Display results in a machine & human readable table
  console.log(colors.bold('CHAIN'.padEnd(12) + 'STATUS'.padEnd(18) + 'BALANCE'.padEnd(24) + 'RECOMMENDATION'));
  console.log('-'.repeat(70));

  let anyInsufficient = false;
  for (const res of results) {
    let statusColor = colors.green;
    if (res.status === 'INSUFFICIENT') {
      statusColor = colors.red;
      anyInsufficient = true;
    } else if (res.status !== 'SUFFICIENT') {
      statusColor = colors.yellow;
    }

    const balanceStr = res.status === 'SUFFICIENT' ? 
      `✓ ${Web3.utils.fromWei(res.currentBalance, 'ether').substring(0,6)}` : 
      `✗ Below min`;

    console.log(
      res.chain.padEnd(12) +
      statusColor(res.status.padEnd(18)) +
      balanceStr.padEnd(24) +
      (res.routeRecommendation || 'CHECK_ERROR')
    );
  }

  // Generate a machine-readable summary for integration
  const auditSummary = {
    timestamp: new Date().toISOString(),
    audit_scope: 'owner_settlement_gas_reserves',
    results: results,
    overall_system_status: anyInsufficient ? 'BLOCKED' : 'OPERATIONAL',
    required_action: anyInsufficient ? 
      'Fund BSC or TRON settlement wallet to minimum threshold' : 
      'All routes approved for autonomous operation'
  };

  // This summary can be consumed by ExternalGatewayManager
  console.log(colors.bold.cyan('\n📋 Agent Audit Summary (JSON):'));
  console.log(JSON.stringify(auditSummary, null, 2));

  // Exit code for CI/automation (0=OK, 1=BLOCKED)
  process.exit(anyInsufficient ? 1 : 0);
}

// ===== EXECUTE AGENT =====
runGasAudit().catch(console.error);
```

**To use this agent:**
1. **Install dependencies**: `npm install web3 tronweb colors`
2. **Set .env variables**: Add `BSC_SETTLEMENT_WALLET=0x...`, `TRON_SETTLEMENT_WALLET=TX...`, etc.
3. **Run**: `node scripts/check-gas-balance.mjs`
4. **Integrate**: Call this script from `ExternalGatewayManager.initiateAutoSettlement()` and block crypto routes if exit code = 1.

#### **B. The "M2M Commerce" Instruction Manifest**

Every autonomous financial action should generate a **machine-readable manifest** (like an NFT for the action itself). This enables the future where agents audit other agents.

Add this to your `CryptoGateway.mjs` when generating instruction files:

```javascript
// When creating a withdrawal instruction, also create a manifest
function generateActionManifest(withdrawalParams, instructionPath) {
  const manifest = {
    standard: "swarm-ai-action-001",
    type: "CRYPTO_SETTLEMENT",
    timestamp: new Date().toISOString(),
    owner: process.env.OWNER_ID,
    target_provider: withdrawalParams.provider,
    intent: {
      asset: withdrawalParams.coin,
      amount: withdrawalParams.amount,
      network: withdrawalParams.network,
      destination: withdrawalParams.address
    },
    proof: {
      instruction_file: instructionPath,
      ledger_entry_id: null, // To be filled after ledger write
      gas_check_passed: true // From check-gas-balance audit
    },
    permissions: {
      autonomous_execution: true,
      requires_human_approval: false,
      max_retries: 3
    }
  };
  
  // Save alongside instruction file
  const manifestPath = instructionPath.replace('.json', '_manifest.json');
  fs.writeFileSync(manifestPath, JSON.stringify(manifest, null, 2));
  return manifest;
}
```
