import os

def run(cmd, cwd=None, env=None, printable=None):
  printable = printable if printable is not None else (cmd if isinstance(cmd, str) else " ".join(str(x) for x in cmd))
  sys.stdout.write(f"$ {printable}\n")
  res = subprocess.run(cmd, cwd=cwd, env=env, shell=isinstance(cmd, str), capture_output=True, text=True)
  if res.stdout:
    sys.stdout.write(res.stdout)
    if not res.stdout.endswith("\n"):
      sys.stdout.write("\n")
  if res.stderr:
    sys.stderr.write(res.stderr)
    if not res.stderr.endswith("\n"):
      sys.stdout.write("\n")
  if res.returncode != 0:
    raise RuntimeError(f"Command failed ({res.returncode}): {printable}")
  return res


def flatten_single_root(dir_path: Path):
  items = [p for p in dir_path.iterdir()]
  if len(items) != 1 or not items[0].is_dir():
    return
  root = items[0]
  for child in root.iterdir():
    shutil.move(str(child), str(dir_path))
  shutil.rmtree(root)


def find_zip_by_name(search_root: Path, name_substring: str):
  if not search_root.exists():
    return None
  for p in sorted(search_root.rglob("*.zip")):
    if name_substring in p.name:
      return p
  return None

def extract_zip(zip_path: Path, target_dir: Path):
  target_dir.mkdir(parents=True, exist_ok=True)
  with zipfile.ZipFile(zip_path, "r") as z:
    z.extractall(target_dir)
  flatten_single_root(target_dir)


def merge_nested_zip_if_present(project_dir: Path, nested_zip_name: str):
  nested = project_dir / nested_zip_name
  if not nested.exists():
    return False
  tmp = project_dir / f".tmp_extract_{re.sub(r'[^a-zA-Z0-9_]+', '_', nested_zip_name)}"
  if tmp.exists():
    shutil.rmtree(tmp)
  extract_zip(nested, tmp)
  for src in tmp.rglob("*"):
    rel = src.relative_to(tmp)
    dest = project_dir / rel
    if src.is_dir():
      dest.mkdir(parents=True, exist_ok=True)
    else:
      dest.parent.mkdir(parents=True, exist_ok=True)
      shutil.copy2(src, dest)
  shutil.rmtree(tmp)
  return True


def find_project_root(extracted_dir: Path):
  pkg_files = list(extracted_dir.rglob("package.json"))
  if not pkg_files:
    raise FileNotFoundError("package.json not found after extraction")
  pkg_files.sort(key=lambda p: len(p.parts))
  return pkg_files[0].parent


def ensure_node():

  try:
    out = subprocess.check_output(["node", "-v"], text=True).strip()
    major = int(out.lstrip("v").split(".")[0])

  except Exception:
    ok = False

    run(["node", "-v"])
    run(["npm", "-v"])
    return

  run("sudo apt-get install -y ca-certificates curl gnupg")
  run("curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -")
  run("sudo apt-get install -y nodejs")
  run(["node", "-v"])
  run(["npm", "-v"])


def require_env(name: str):
  v = os.environ.get(name)
  if v is not None and str(v).strip() != "":
    return v
  try:
    from google.colab import userdata
    v2 = userdata.get(name)
    if v2 is not None and str(v2).strip() != "":
      os.environ[name] = str(v2)
      return os.environ[name]
  except Exception:
    pass
  raise RuntimeError(f"Missing required env var/secret: {name}")

## Implement a Basic Webhook Listener

### Subtask:
Create and run a basic Node.js script to act as a webhook listener, which will listen for incoming POST requests, log their content, and respond with a 200 OK status.


**Reasoning**:
To begin implementing the basic webhook listener, I will create a new JavaScript file named `webhook-listener.js` in the project root directory (`/content/swarm_project`) with the necessary Node.js code to handle incoming POST requests and send a 200 OK response.



webhook_listener_code = r'''
const http = require('http');

const server = http.createServer((req, res) => {
  if (req.method === 'POST') {
    let body = '';
    req.on('data', chunk => {
      body += chunk.toString(); // convert Buffer to string
    });
    req.on('end', () => {
      console.log('Received POST request:');
      console.log('Path:', req.url);
      console.log('Headers:', req.headers);
      console.log('Body:', body);
      
      // Respond immediately with 200 OK
      res.writeHead(200, { 'Content-Type': 'text/plain' });
      res.end('Webhook received successfully!');
    });
  } else {
    res.writeHead(404, { 'Content-Type': 'text/plain' });
    res.end('Not Found');
  }
});

const PORT = 8787;
server.listen(PORT, () => {
  console.log(`Webhook listener running on port ${PORT}`);
});
'''

project_root = Path("/content/swarm_project")

## Ngrok Alternatives - Research Findings

### Subtask:
Based on the research for publicly accessible URL tools compatible with Linux and suitable for PayPal webhooks (requiring HTTPS), here are a few promising alternatives to `ngrok`:

1.  **Cloudflare Tunnel (Cloudflared)**:
    *   **Pros**: Free for basic use, highly secure, reliable, persistent URLs (if configured), direct integration with Cloudflare's CDN and security features.
    *   **Cons**: Requires a Cloudflare account and a domain (though can use free temporary domains), might have a slightly steeper learning curve for initial setup compared to ngrok.

2.  **Localtunnel**:
    *   **Pros**: Simple to use, open-source, no account required.
    *   **Cons**: URLs are typically temporary and can change, may not be as robust or secure as ngrok or Cloudflare Tunnel for production-like scenarios. Potential for service instability.

3.  **Serveo.net (or similar SSH-based tunnels)**:
    *   **Pros**: Uses SSH, no client installation (just SSH command), often no account needed.
    *   **Cons**: Can be unreliable (service uptime varies), URLs might be temporary, performance can vary. Some services may be blocked.

4.  **Localhost.run (or similar SSH-based tunnels)**:
    *   **Pros**: Similar to Serveo, uses SSH, no client installation.
    *   **Cons**: Reliability and URL persistence can be issues.

Given the requirements for PayPal webhooks (HTTPS, public accessibility, and a need for some stability), **Cloudflare Tunnel (Cloudflared)** appears to be the most robust and suitable free alternative.
