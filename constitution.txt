## 🔐 **2. CRYPTOGRAPHIC INTEGRITY FRAMEWORK**

**File:** `/constitution/security/integrity-manager.mjs`

```javascript
import crypto from 'crypto';
import fs from 'fs/promises';
import path from 'path';

class ConstitutionIntegrityManager {
  constructor() {
    this.constitutionPath = path.join(process.cwd(), 'constitution', 'swarm.constitution.json');
    this.ownerPublicKeyPath = path.join(process.cwd(), 'constitution', 'owner_public.key');
    this.masterAgentPublicKeyPath = path.join(process.cwd(), 'constitution', 'masteragent_public.key');
  }

  async generateConstitutionIntegrity() {
    // 1. Read and canonicalize constitution
    const constitution = await fs.readFile(this.constitutionPath, 'utf-8');
    const canonicalJson = JSON.stringify(JSON.parse(constitution), null, 2);
    
    // 2. Generate SHA-256 hash
    const hash = crypto.createHash('sha256').update(canonicalJson).digest('hex');
    await fs.writeFile(path.join(process.cwd(), 'constitution', 'constitution.hash.txt'), hash);
    
    // 3. Generate HMAC with audit secret
    const hmacSecret = process.env.AUDIT_HMAC_SECRET;
    if (!hmacSecret) throw new Error('AUDIT_HMAC_SECRET not set');
    const hmac = crypto.createHmac('sha256', hmacSecret).update(canonicalJson).digest('hex');
    await fs.writeFile(path.join(process.cwd(), 'constitution', 'constitution.hmac.txt'), hmac);
    
    // 4. Generate signatures (these would be generated offline in production)
    const signatures = {
      generated_at: new Date().toISOString(),
      constitution_hash: hash,
      hmac: hmac,
      note: "Owner and MasterAgent signatures must be generated offline and placed in owner.signature and masteragent.signature files"
    };
    
    console.log('✅ Constitution integrity files generated');
    console.log(`📋 Hash: ${hash}`);
    console.log('🔐 HMAC generated with AUDIT_HMAC_SECRET');
    console.log('⚠️  IMPORTANT: Generate offline signatures and place in constitution/ directory');
    
    return signatures;
  }

  async validateConstitutionAtRuntime() {
    // Runtime validation before any autonomous action
    const constitution = await fs.readFile(this.constitutionPath, 'utf-8');
    const canonicalJson = JSON.stringify(JSON.parse(constitution), null, 2);
    
    // 1. Verify hash
    const currentHash = crypto.createHash('sha256').update(canonicalJson).digest('hex');
    const storedHash = await fs.readFile(
      path.join(process.cwd(), 'constitution', 'constitution.hash.txt'), 'utf-8'
    ).catch(() => null);
    
    if (currentHash !== storedHash?.trim()) {
      throw new Error('CONSTITUTION_TAMPERED: Hash mismatch');
    }
    
    // 2. Verify HMAC
    const hmacSecret = process.env.AUDIT_HMAC_SECRET;
    if (!hmacSecret) throw new Error('AUDIT_HMAC_SECRET not set');
    
    const currentHmac = crypto.createHmac('sha256', hmacSecret).update(canonicalJson).digest('hex');
    const storedHmac = await fs.readFile(
      path.join(process.cwd(), 'constitution', 'constitution.hmac.txt'), 'utf-8'
    ).catch(() => null);
    
    if (currentHmac !== storedHmac?.trim()) {
      throw new Error('CONSTITUTION_TAMPERED: HMAC mismatch');
    }
    
    // 3. Verify signatures (placeholder - would verify against public keys)
    const ownerSignatureExists = await fs.access(
      path.join(process.cwd(), 'constitution', 'owner.signature')
    ).then(() => true).catch(() => false);
    
    const masterAgentSignatureExists = await fs.access(
      path.join(process.cwd(), 'constitution', 'masteragent.signature')
    ).then(() => true).catch(() => false);
    
    if (!ownerSignatureExists || !masterAgentSignatureExists) {
      console.warn('⚠️  Constitution signatures not verified - running in degraded trust mode');
    }
    
    return {
      valid: true,
      hash: currentHash,
      validated_at: new Date().toISOString()
    };
  }
}

export default ConstitutionIntegrityManager;
```

```javascript
class PreActionValidator {
  constructor(constitution) {
    this.constitution = constitution;
  }

  validateIntent(intent, context) {
    const violations = [];
    
    // 1. Check against allowed intents
    if (!this.constitution.goal_mission_policy.allowed_intents.includes(intent.type)) {
      violations.push({
        level: 'CRITICAL',
        code: 'INTENT_NOT_ALLOWED',
        message: `Intent type ${intent.type} is not in allowed intents`
      });
    }
    
    // 2. Check forbidden intents
    if (this.constitution.goal_mission_policy.forbidden_intents.includes(intent.type)) {
      violations.push({
        level: 'CRITICAL',
        code: 'INTENT_FORBIDDEN',
        message: `Intent type ${intent.type} is explicitly forbidden`
      });
    }
    
    // 3. Check owner baseline invariants
    this.checkOwnerBaseline(intent, context, violations);
    
    // 4. Check financial constraints
    this.checkFinancialConstraints(intent, context, violations);
    
    // 5. Check jurisdiction constraints
    this.checkJurisdictionConstraints(intent, context, violations);
    
    return {
      valid: violations.length === 0,
      violations,
      requires_human_approval: violations.some(v => v.level === 'MEDIUM') || 
                               intent.amount_usd > this.constitution.human_in_the_loop_escalation.risk_tiered_approvals.tier_2_threshold
    };
  }

  checkOwnerBaseline(intent, context, violations) {
    const baseline = this.constitution.foundational_invariants.owner_baseline;
    
    // Example: Check for gambling patterns
    if (intent.description?.toLowerCase().includes('bet') || 
        intent.description?.toLowerCase().includes('casino')) {
      violations.push({
        level: 'CRITICAL',
        code: 'BASELINE_VIOLATION',
        rule: 'NO_GAMBLING',
        message: 'Transaction appears to be gambling-related'
      });
    }
    
    // Add more baseline checks here
  }

  checkFinancialConstraints(intent, context, violations) {
    const constraints = this.constitution.goal_mission_policy.constraints;
    
    if (intent.amount_usd > constraints.max_settlement_per_transaction_usd) {

        code: 'FINANCIAL_CONSTRAINT_VIOLATION',

        limit: constraints.max_settlement_per_transaction_usd,
        actual: intent.amount_usd,
        message: `Transaction amount exceeds per-transaction limit`
      });
    }
  }
}
```

```javascript
class SafetyMonitor {
  constructor(constitution) {
    this.constitution = constitution;
    this.anomalyDetector = new AnomalyDetector();
    this.circuitBreakers = new Map();
  }

  async monitorActionExecution(action, result) {
    const safetyEnvelopes = this.constitution.model_risk_safety.safety_envelopes;
    
    // 1. Check financial safety envelope
    if (result.actual_loss_usd > safetyEnvelopes.financial_safety.max_loss_tolerance_percent * action.expected_amount_usd) {
      await this.triggerCircuitBreaker('FINANCIAL_SAFETY_VIOLATION', {
        action,
        result,
        threshold: safetyEnvelopes.financial_safety.max_loss_tolerance_percent
      });
    }
    
    // 2. Detect anomalies
    const anomalies = await this.anomalyDetector.detect(action, result);
    if (anomalies.length > 0) {
      await this.escalateToHuman('ANOMALY_DETECTED', { action, anomalies });
    }
    
    // 3. Update drift metrics
    await this.updateModelDriftMetrics(action, result);
  }

  async triggerCircuitBreaker(reason, data) {
    const killSwitches = this.constitution.incident_response_recovery.kill_switches;
    
    // Activate global kill switch if needed
    if (killSwitches.global_kill_switch === 'ENABLED') {
      await this.activateGlobalKillSwitch(reason, data);
    }
    
    // Log for audit
    await this.logIncident({
      type: 'CIRCUIT_BREAKER_TRIGGERED',
      reason,
      data,
      timestamp: new Date().toISOString(),
      automatic_response: 'SYSTEM_HALTED'
    });
    
    // Notify owner
    await this.notifyOwner(`🚨 CIRCUIT BREAKER ACTIVATED: ${reason}`);
  }
}
```

```javascript
class HumanEscalationGateway {
  constructor(constitution) {
    this.constitution = constitution;
    this.pendingApprovals = new Map();
  }

  async requestApproval(action, context, riskTier) {
    const approvalId = crypto.randomBytes(16).toString('hex');
    
    // Store approval request
    this.pendingApprovals.set(approvalId, {
      action,
      context,
      riskTier,
      requestedAt: new Date().toISOString(),
      status: 'PENDING'
    });
    
    // Determine escalation method based on risk tier
    const escalationConfig = this.constitution.human_in_the_loop_escalation;
    
    switch(riskTier) {
      case 'TIER_1_LOW':
        // Just notify
        await this.sendNotification(action);
        return { approved: true, approvalId };
        
      case 'TIER_2_MEDIUM':
        // Require explicit approval
        const approval = await this.requestExplicitApproval(action, approvalId);
        return approval;
        
      case 'TIER_3_HIGH':
        // Immediate escalation + hold
        await this.immediateEscalation(action, approvalId);
        return { approved: false, approvalId, status: 'ESCALATED' };
    }
  }

  async requestExplicitApproval(action, approvalId) {
    const consentUX = this.constitution.human_in_the_loop_escalation.consent_ux;
    
    // Generate OTP for verification
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    
    // Send via configured modalities
    if (consentUX.modality.includes('MOBILE_PUSH')) {
      await this.sendMobilePush(action, otp);
    }
    
    if (consentUX.modality.includes('EMAIL_OTP')) {
      await this.sendEmailOTP(action, otp);
    }
    
    // Wait for response with timeout
    return new Promise((resolve) => {
      const timeout = setTimeout(() => {
        const defaultAction = consentUX.default_action_on_timeout;
        resolve({
          approved: defaultAction === 'APPROVE',
          approvalId,
          reason: defaultAction === 'ABORT_AND_HOLD' ? 'TIMEOUT' : 'DEFAULT_APPROVAL'
        });
      }, consentUX.timeout_seconds * 1000);
      
      // In reality, would wait for webhook/response
      // For now, simulate approval
      clearTimeout(timeout);
      resolve({ approved: true, approvalId });
    });
  }
}
```

```javascript
import ConstitutionIntegrityManager from './integrity-manager.mjs';
import PreActionValidator from './pre-action-validator.mjs';
import SafetyMonitor from './safety-monitor.mjs';
import HumanEscalationGateway from './human-escalation-gateway.mjs';

class ConstitutionEnforcedSwarm {
  constructor() {
    this.integrityManager = new ConstitutionIntegrityManager();
    this.constitution = null;
    this.validator = null;
    this.safetyMonitor = null;
    this.escalationGateway = null;
  }

  async initialize() {
    // 1. Validate constitution integrity at startup
    const integrity = await this.integrityManager.validateConstitutionAtRuntime();
    console.log(`✅ Constitution validated: ${integrity.hash.substring(0, 16)}...`);
    
    // 2. Load constitution
    const constitution = await this.loadConstitution();
    this.constitution = constitution;
    
    // 3. Initialize enforcement components
    this.validator = new PreActionValidator(constitution);
    this.safetyMonitor = new SafetyMonitor(constitution);
    this.escalationGateway = new HumanEscalationGateway(constitution);
    
    console.log('🏛️  Constitution Enforcement Layer Active');
  }

  async executeSettlementWithConstitution(intent) {
    // STEP 1: Pre-action validation
    const validation = this.validator.validateIntent(intent, {});
    
    if (!validation.valid) {
      console.error('❌ Constitution validation failed:', validation.violations);
      await this.handleValidationFailure(validation.violations);
      return { success: false, reason: 'CONSTITUTION_VIOLATION' };
    }
    
    // STEP 2: Check for human approval required
    if (validation.requires_human_approval) {
      const approval = await this.escalationGateway.requestApproval(
        intent, 
        {}, 
        'TIER_2_MEDIUM'
      );
      
      if (!approval.approved) {
        return { success: false, reason: 'HUMAN_DENIAL', approvalId: approval.approvalId };

      // STEP 4: Post-action safety monitoring
      await this.safetyMonitor.monitorActionExecution(intent, result);
      
      // STEP 5: Audit trail

    }
```
